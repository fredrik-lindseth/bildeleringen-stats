# Varsler Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Overvåk sted+filtre i flex og varsle (browser + varsel-relay) når en bil blir ledig en sammenhengende blokk ≥ terskel.

**Architecture:** Pollingen skjer i nettleseren (dele.no-token bor der). `background.js` vekkes av `alarms`, søker per aktiv overvåkning, kjører rene funksjoner i `lib/watch.js` (`detectHits` + `reconcile`) for deteksjon og de-dup, og fyrer ett browser-varsel + én POST til en dedikert backend på serveren som relayer til Signal. Backend er eget delsystem (egen plan i backend-repoet, delegeres).

**Tech Stack:** Vanilla ES modules (unntatt `content.js`), `node --test`, ingen build-step, ingen npm-avhengigheter.

## Global Constraints

- **Ingen build-step, ingen npm-pakker.** Vanilla JS/HTML/CSS, ES modules overalt unntatt `content.js` (klassisk script).
- **Alle farger via CSS `:root`-variabler.** Ingen hardkodet hex utenfor `:root`. Mørk modus påkrevd (`prefers-color-scheme` + dark override for nye variabler).
- **Norsk UI-tekst.** Alt brukeren ser på norsk. `console.log` kan være engelsk.
- **Rene stats/deteksjons-funksjoner:** `lib/watch.js` har ingen sideeffekter, ingen browser-API, ingen storage.
- **Cross-browser:** `typeof browser !== "undefined" ? browser : chrome`. Bruk `browserAPI`/`storage` fra `lib/browser-polyfill.js` i moduler.
- **Tider er naiv lokal ISO = Europe/Oslo veggklokke.** Bruk `toLocalISO`/`parseLocalISO` fra `lib/format.js`. Aldri parse som UTC. `deepLink`-tider sendes urørt (ingen URL-enkoding), ellers "Invalid date" i dele.no-appen.
- **De-dup-nøkkel = `` `${carId}|${toLocalISO(blockEnd)}` ``.** Blokkslutt, ikke blokkstart (blokkstart drifter etter vindusstart).
- **`background.js` er eneste skriver på `storage.local.watches`.** UI muterer via meldinger.
- **Firefox-innlasting:** UI-endringer (HTML/JS/CSS på utvidelses-sider) krever bare refresh av fanen. `background.js`/`manifest.json`/`content.js`-endringer krever reinstall (`install_extension`) eller "Reload" i `about:debugging`. Kjør via `firefox-devtools`-MCP-en mot dens dedikerte profil.

```js
import { toLocalISO, parseLocalISO } from "./format.js";
import { freeBlocks, blockMinutes, matchesFilters, withinDistance } from "./flex.js";

// window = { start: Date, end: Date } — det effektive vinduet polleren søkte med.
// Gis eksplisitt; funksjonen leser aldri watch.earliestStart selv.
export function detectHits(searchResult, watch, window) {
  const cars = (searchResult.cars || [])
    .filter((c) => matchesFilters(c, watch.filters))
    .filter((c) => withinDistance(c, watch.filters));
  const minMinutes = watch.minHours * 60;
  const hits = [];
  for (const c of cars) {
    for (const b of freeBlocks(c.availability, window.start, window.end)) {
      const mins = blockMinutes(b);
      if (mins < minMinutes) continue;
      hits.push({
        carId: c.carId,
        carNumber: c.carNumber,
        model: c.model,
        category: c.carCategory,
        location: { name: c.location?.name, distance: c.location?.distance },
        blockStart: b.start,
        blockEnd: b.end,
        blockHours: mins / 60,
      });
    }
  }
  return hits;
}
```

- [ ] **Step 3: Skriv `reconcile`**

```js
function hitKey(hit) {
  return `${hit.carId}|${toLocalISO(hit.blockEnd)}`;
}

// Ren de-dup + rydding + relay-retry. Nøkkel på blokkslutt (stabil på tvers av
// polls). notified inn er watchens hovedbok; now er en Date.
export function reconcile(hits, notified, now) {
  const prev = notified || {};
  const next = {};
  // Behold eksisterende nøkler som ikke er passert.
  for (const [key, entry] of Object.entries(prev)) {
    const endISO = key.slice(key.indexOf("|") + 1);
    if (parseLocalISO(endISO) > now) next[key] = entry;
  }
  const browserHits = [];
  const relayHits = [];
  for (const hit of hits) {
    const key = hitKey(hit);
    const existing = prev[key];
    if (!existing) {
      browserHits.push(hit);
      relayHits.push(hit);
      next[key] = { notifiedAt: toLocalISO(now), relayed: false };
    } else {
      if (existing.relayed === false) relayHits.push(hit);
      next[key] = existing; // bevar relayed-status
    }
  }
  return { browserHits, relayHits, notified: next };
}
```

```js
      sendResponse({ watches, settings })
    );
    return true;

    return true;

    saveSettings(message.settings).then((settings) => sendResponse({ settings }));
```

- [ ] **Step 2: Skriv `pollWatches`**

```js
async function pollWatches() {
  const now = new Date();
  const settings = await getSettings();
  let watches = await getWatches();

  // Utløp
  watches = watches.map((w) =>
    parseLocalISO(w.latestEnd) <= now ? { ...w, active: false } : w);

  const token = await getToken();
  const results = [];
  if (!token) {
    await setWatches(watches);
    console.log("[bg] pollWatches: ingen token, hopper over");
    return { results, noAuth: true };
  }

  for (let i = 0; i < watches.length; i++) {
    const w = watches[i];
    if (!w.active || parseLocalISO(w.latestEnd) <= now) continue;
    const start = new Date(Math.max(parseLocalISO(w.earliestStart).getTime(), now.getTime()));
    const end = parseLocalISO(w.latestEnd);
    let result;
    try {
      result = await searchAvailability(w.point, start, end, token);
    } catch (e) {
      if (e.message === "AUTH_EXPIRED") {
        await storage.set({ authToken: null });
        currentAuth = null;
        console.log("[bg] pollWatches: 401, token tømt");
        break;
      }
      console.warn("[bg] pollWatches søkefeil:", e.message);
      continue;
    }
    const hits = detectHits(result, w, { start, end });
    const rec = reconcile(hits, w.notified, now);
    watches[i] = { ...w, notified: rec.notified, lastChecked: now.toISOString() };
    results.push({ watchLabel: w.label, watch: watches[i], settings,
                   browserHits: rec.browserHits, relayHits: rec.relayHits });
  }
  await setWatches(watches);
  return { results, noAuth: false };
}
```

## Task 7: Dispatcher (browser-varsel + relay-POST + relayed-merking)

**Files:**
- Modify: `flex/background.js`

**Interfaces:**
- Consumes: `pollWatches`-resultat `{ watchLabel, watch, settings, browserHits, relayHits }[]`.
- Produces: `dispatchAll(results)` som fyrer ett browser-varsel per watch med `browserHits`, én POST per watch med `relayHits`, og setter `relayed: true` på relaykede nøkler (via `setWatches`).

- [ ] **Step 1: Skriv dispatcher**

Erstatt stub-en fra Task 6:
```js
import { toLocalISO } from "./lib/format.js";

function summaryText(watch, hits) {
  if (hits.length === 1) {
    const h = hits[0];
    return `${h.model} ${h.carNumber} ledig ${Math.round(h.blockHours)} t ved ${h.location.name}`;
  }
  return `${hits.length} biler ledige ≥ ${watch.minHours} t ved ${watch.placeName}`;
}

function toPayloadHit(h) {
  const start = toLocalISO(h.blockStart);
  const end = toLocalISO(h.blockEnd);
  return {
    car: { model: h.model, number: h.carNumber, location: h.location.name, distanceM: h.location.distance },
    free: { start, end, hours: Math.round(h.blockHours) },
    // Tider urørt, ingen URL-enkoding (kolon-gotcha).
    deepLink: `https://app.dele.no/findcar?start=${start}&end=${end}`,
  };
}

async function relay(settings, watch, relayHits) {
  const res = await fetch(`${settings.backendUrl}/api/notify`, {
    method: "POST",
    headers: { "content-type": "application/json", authorization: `Bearer ${settings.backendToken}` },
    body: JSON.stringify({
      watchLabel: watch.label, place: watch.placeName,
      hits: relayHits.map(toPayloadHit),
    }),
  });
  if (!res.ok) throw new Error(`relay ${res.status}`);
}

async function dispatchAll(results) {
  let watches = await getWatches();
  for (const r of results) {
    if (r.browserHits.length) {
      browserAPI.notifications.create(`watch-${r.watch.id}-${Date.now()}`, {
        type: "basic",
        iconUrl: browserAPI.runtime.getURL("icons/icon-48.png"),
        title: "Bildeleringen: bil ledig",
        message: summaryText(r.watch, r.browserHits),
      });
    }
    if (r.relayHits.length && r.settings.backendUrl && r.settings.backendToken) {
      try {
        await relay(r.settings, r.watch, r.relayHits);
        // Merk relaykede nøkler relayed=true i den ferske lista.
        watches = watches.map((w) => {
          if (w.id !== r.watch.id) return w;
          const notified = { ...w.notified };
          for (const h of r.relayHits) {
            const key = `${h.carId}|${toLocalISO(h.blockEnd)}`;
            if (notified[key]) notified[key] = { ...notified[key], relayed: true };
          }
          return { ...w, notified };
        });
      } catch (e) {
        console.warn("[bg] relay feilet, retry neste runde:", e.message);
      }
    }
  }
  await setWatches(watches);
}
```

```js
const api = typeof browser !== "undefined" ? browser : chrome;

function fmtWindow(w) {
  return `${w.earliestStart.slice(0,16).replace("T"," ")} – ${w.latestEnd.slice(0,16).replace("T"," ")}`;

function status(w) {

  if (new Date(w.latestEnd) <= new Date()) return "utløpt";
  return w.active ? "aktiv" : "av";

async function render() {
  const { watches, settings } = await api.runtime.sendMessage({ type: "GET_WATCHES" });
  const ul = document.getElementById("watch-list");
  ul.innerHTML = "";
  if (!watches.length) { ul.innerHTML = "<li>Ingen overvåkninger ennå.</li>"; }
  for (const w of watches) {

    const checked = w.active ? "checked" : "";
    li.innerHTML =
      `<strong>${w.label}</strong> · ${status(w)}<br>` +
```
